   Synopsis: 
   + Hash-Map = a key-value store that uses an array and a hashing function to save and retrieve values.
   + Key = identifier given to a value for later retrieval 
   + Hash-Function = a function that takes some input and returns a number
   + Compression function = a function that transforms its inputs into some smaller range of possible inputs 
   
   + Recipe for saving to a hash table: 
   - Take the key and plug it into the hash-function, returning the hash code
   - Modulo the hash code by the length of the underlying array, returning the array index
   - Check if the array at the given index is empty, if so save the value/key pair to that index
   - if the array is full at that index continue to the next possible position depending on your collision strategy
   
   + Recipe for retrieving from a hash table:
   - Take the key and plug it into the hash function, getting the hash code. 
   - Modulo that hash code by the length of the underlying array, return the index
   - Check if the array at that index has contents, if so check the key saved there. 
   - If the key matches the one you're looking for, return the value. 
   - If the keys don't match, continue to the next position depending on your collision strategy. 
   + There are two well known collision strategies
   - Separate chaining - where each index of the hash-map points to a linked list
   - Open addressing - where if the index of the map is occupied, it moves to another spot in the list until an empty spot is found

const Node = require('./Node');
class HashMap {
  constructor(size = 0) {
    this.hashmap = new Array(size)
      .fill(null)
      .map(() => new LinkedList());
  }

  hash(key) {
    let hashCode = 0;
    for (let i = 0; i < key.length; i++) {
      hashCode += hashCode + key.charCodeAt(i);
    }
    return hashCode % this.hashmap.length;
  }

  assign(key, value) {
    const arrayIndex = this.hash(key);
    const linkedList = this.hashmap[arrayIndex];
    if (linkedList.head === null) {
      linkedList.addToHead({ key, value });
      return;
    }
    let current = linkedList.head;
    while (current) {
      if (current.data.key === key) {
        current.data = { key, value };
      }
      if (!current.next) {
        current.next = new Node({ key, value });
        break;
      }
      current = current.next;
    }
  }

  retrieve(key) {
    const arrayIndex = this.hash(key);
    let current = this.hashmap[arrayIndex].head;
    while (current) {
      if (current.data.key === key) {
        return current.data.value;
      }
      current = current.getNextNode();
    }
    return null;
  }
}

module.exports = HashMap;

class Stack {
   constructor(maxSize = Infinity) {
      this.stack = new LinkedList();
      this.maxSize = maxSize;
      this.size = 0;
   }
   hasRoom() {
      if (this.size < this.maxSize) {
         return true;
      } else {
         return false;
      }
   }
   isEmpty() {
      return this.size === 0;
   }
   push(value) {
      if (this.hasRoom()) {
         this.stack.addToHead(value);
         ++this.size;
      } else {
         throw new Error("Stack is full.");
      }
   }
   pop() {
      if (!this.isEmpty()) {
         const value = this.stack.removeHead();
         --this.size;
         return value;
      } else {
         console.log('Stack is empty!');
      }
   }
   peek() {
      if (!this.isEmpty()) {
         return this.stack.head.data;
      } else {
         return null;
      }
   }
}

Queues
   + Many ways to implement, often uses singly linked lists
   + It enqueus and dequeues items from a linked list. 
   + Works like a line at a grocery store. 
   + Enqueue - adds data to the tail of the list
   + Dequeue - removes the head of the list
   + Peek - reveals data at the head of the list
   + FIFO
   + usually relinking or reordering the nodes of a list is restricted in some specific way
   + there is often a length limit in a bounded queue
   + Adding to a full list is overflow, removing from an empty list is underflow
  
Examples 
class Queue {
   constructor(maxSize = Infinity) {
      this.queue = new LinkedList();
      this.maxSize = maxSize;
      this.size = 0;
   }
   isEmpty() {
      return this.size === 0;
   }
   hasRoom() {
      return this.size < this.maxSize;
   }
   enqueue(data) {
      if (this.hasRoom()) {
         this.queue.addToTail(data);
         ++this.size;
      } else {
         throw new Error("Queue is full!");
      }
   }
   dequeue() {
      if (!this.isEmpty()) {
         const data = this.queue.removeHead();
         --this.size;
         return data;
      } else {
         throw new Error("Queue is empty!");
      }
   }
}

node2Prev.setNextNode(node1);

  node1.setNextNode(node2.getNextNode());
  node2.setNextNode(temp);

class LinkedList {
   constructor(data) {
      this.head = null;
   }
   addToHead(data) {
      const newHead = new Node(data); // make new node
      const currentHead = this.head; // store node we are replacing
      this.head = newHead;          // replace head with new node
      if (currentHead !== null) {
         this.head.setNextNode(currentHead); // check if old head is connected, if so set new head to link to old head.
      }
   }
   // add to end of list
   addToTail(data) {
      let tail = this.head; // start at front
      if (tail === null) { // if front is empty then 
         this.head = new Node(data); // make new head node
      } else {
         while (tail.getNextNode() !== null) { // while loop, condition is check the next key of current tail node isn't null
            tail = tail.getNextNode() // set the current tail node to the next node if not null
         }
         tail.setNextNode(new Node(data)); // once the condition exits we should be at the end of the list so make the new tail node
      }
   }
   // remove the current head, or first node
   removeHead() {
      const removedHead = this.head; // store head to be removed
      if (!removedHead) {           // check if the head exists, if not stop the function
         return;
      }
      this.head = removedHead.getNextNode(); // set head to the next node in line
      return removedHead.data;       // return the removed heads data
   }
   // print entire list from head to tail
   printList() {
      let currentNode = this.head; // temp variable that changes as we go down the list
      let output = '<head> '; // start of the formatted string
      while (currentNode) { // loop while the currentNode exists, isn't null
         output += currentNode.data + ' '; // add the currentNodes data key to the format string
         currentNode = currentNode.getNextNode(); // change currentNode to the next node in the list, next key of currentNode
      }
      output += '<tail>'; // once the loop finishes that should be the end of the list so add tail to the format string end
      console.log(output); // output the complete list to the console.
   }
}
